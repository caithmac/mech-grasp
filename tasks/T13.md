# T13 - Are the SAE chemistry features USED? Causal ablation and steering of layer-6 features

Question: T08b found layer-6 SAE latents that encode chemistry in context (e.g. "this C carries an OH").
Correlation is not use. Remove a feature from the residual stream at layer 6 and ask whether the
information disappears downstream (layers 7-12 and the molecule logP read-out) more than an equal-size
random edit does. Then ADD the feature to atoms that lack it (steering = sufficiency).
ALL row bookkeeping is done by mechlib.edits (tested) - do not write your own mapping code.

def run_feature(j, concept, sign, fam):        # returns a JSON-able dict; wrap in utils.memo(f"t13_v3_{j}", ...)
  z = Zc[:, j].toarray().ravel(); tok = int(np.bincount(A.tokens[z > 0]).argmax()); T = A.tokens == tok
  c = df[concept].values.astype(int); rng = np.random.default_rng(j)
  within_auroc = roc_auc_score(c[T], z[T])          # from sklearn.metrics
  targets = np.where(T & (z > 0) & (c == 1))[0]; steer_rows = np.where(T & (z == 0) & (c == 0))[0]
  keep at most 300 of each: x = rng.choice(x, 300, replace=False) if len(x) > 300 else x
  P = {L: probes.FixedProbe(A.atom[L], c, g, mask=T) for L in LS}
  gap = {L: P[L].decision(A.atom[L][T & (c == 1)]).mean() - P[L].decision(A.atom[L][T & (c == 0)]).mean() for L in LS}
  pa = edits.prepare(enc, smi, A, targets, layers=[6] + LS)
  Zt = S.encode(pa.clean.atom[6][pa.target])                      # (n_targets, 4096) codes of the targets
  d_abl = -Zt[:, [j]] * RD[j]                                      # (n, 768)
  d_fam = -sum(Zt[:, [f]] * RD[f] for f in fam)
  r = rng.normal(size=d_abl.shape); d_rnd = r / np.linalg.norm(r, axis=1, keepdims=True) * np.linalg.norm(d_abl, axis=1, keepdims=True)
  ps = edits.prepare(enc, smi, A, steer_rows, layers=[6] + LS)
  zmed = float(np.median(z[targets])); d_st = np.tile(zmed * RD[j], (len(steer_rows), 1))
  r = rng.normal(size=d_st.shape); d_str = r / np.linalg.norm(r, axis=1, keepdims=True) * np.linalg.norm(d_st[0])
  For cond, prep, delta in [("ablate", pa, d_abl), ("ablate_fam", pa, d_fam), ("random", pa, d_rnd),
                            ("steer", ps, d_st), ("steer_rand", ps, d_str)]:
     ed = edits.apply(enc, prep, delta, layer=6); cl = prep.clean; t = prep.target; nb = prep.nbr
     per L in LS:
       dP = P[L].decision(ed.atom[L][t]) - P[L].decision(cl.atom[L][t])
       concept_removed = -dP.mean() / gap[L]  (ablation conditions)  or  steer_shift = dP.mean() / gap[L]  (steer conditions)
       se = dP.std() / sqrt(len(dP)) / |gap[L]|
       steer_flip = mean((P[L].decision(cl.atom[L][t]) < 0) & (P[L].decision(ed.atom[L][t]) > 0))   (steer conditions)
       crippen_delta = mean(Q[L].decision(ed.atom[L][t]) - Q[L].decision(cl.atom[L][t]))
       nbr_crippen_delta = same on nb (skip if len(nb) == 0)
       rel_change = mean(||ed.atom[L][t] - cl.atom[L][t]|| / ||cl.atom[L][t]||)
     ablation conditions only, per L in (8, 12): dl = M[L].decision(ed.mean[L]) - M[L].decision(cl.mean[L]);
       dlogp = dl.mean(), dlogp_se, sign_agree = mean(np.sign(dl) == sign); also cls12_rel = mean ||ed.cls[12]-cl.cls[12]|| / ||cl.cls[12]||
